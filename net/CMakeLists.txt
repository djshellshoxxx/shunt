file(GLOB SHUNT_NET_SOURCES CONFIGURE_DEPENDS src/*.cpp)
if(WIN32)
  list(FILTER SHUNT_NET_SOURCES EXCLUDE REGEX "PosixSocket\\.cpp$")

  list(FILTER SHUNT_NET_SOURCES EXCLUDE REGEX "WinSocket\\.cpp$")
endif()

if(WIN32)
  target_link_libraries(shunt_net PUBLIC ws2_32 iphlpapi)
endif()
